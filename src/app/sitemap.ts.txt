import type { MetadataRoute } from "next";

const siteUrl = process.env.NEXT_PUBLIC_SITE_URL ?? "https://www.peonystudio.net";

export default function sitemap(): MetadataRoute.Sitemap {
  return [
    "",
    "/come-iniziare",
    "/percorsi",
    "/pratica",
    "/workshop",
    "/calendario",
    "/peony",
    "/shop",
    "/contatti",
  ].map((path) => ({
    url: `${siteUrl}${path}`,
    lastModified: new Date(),
    changeFrequency: "weekly",
    priority: path === "" ? 1 : 0.8,
  }));
}
